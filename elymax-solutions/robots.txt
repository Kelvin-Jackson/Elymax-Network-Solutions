        // Handle change password form submission
        function handleChangePassword(event) {
            event.preventDefault();
            
            const currentUsername = document.getElementById('current-username').value;
            const currentPassword = document.getElementById('current-password').value;
            const newPassword = document.getElementById('new-password').value;
            const confirmNewPassword = document.getElementById('confirm-new-password').value;
            
            // Validate inputs
            if (newPassword !== confirmNewPassword) {
                document.getElementById('change-password-error').textContent = 'New passwords do not match';
                document.getElementById('change-password-error').classList.remove('hidden');
                return;
            }
            
            if (newPassword.length < 6) {
                document.getElementById('change-password-error').textContent = 'New password must be at least 6 characters long';
                document.getElementById('change-password-error').classList.remove('hidden');
                return;
            }
            
            // Find user and verify current password
            const userIndex = adminCredentials.findIndex(cred => 
                cred.username === currentUsername && cred.password === currentPassword
            );
            
            if (userIndex === -1) {
                document.getElementById('change-password-error').textContent = 'Current password is incorrect';
                document.getElementById('change-password-error').classList.remove('hidden');
                return;
            }
            
            // Update password
            adminCredentials[userIndex].password = newPassword;
            saveCredentials();
            
            // Update current user data in localStorage if it's the logged-in user
            const currentUserData = JSON.parse(localStorage.getItem('adminUserData') || '{}');
            if (currentUserData.username === currentUsername) {
                currentUserData.password = newPassword;
                localStorage.setItem('adminUserData', JSON.stringify(currentUserData));
            }
            
            // Show success message
            document.getElementById('change-password-success').classList.remove('hidden');
            document.getElementById('change-password-error').classList.add('hidden');
            
            // Reset form and close modal after delay
            setTimeout(() => {
                closeChangePasswordModal();
            }, 2000);
        }

        // Handle edit user form submission
        function handleEditUser(event) {
            event.preventDefault();
            
            const originalUsername = document.getElementById('edit-username-original').value;
            const newUsername = document.getElementById('edit-username').value;
            const newPassword = document.getElementById('edit-password').value;
            const newRole = document.getElementById('edit-role').value;
            
            // Get selected permissions
            const permissions = [];
            document.querySelectorAll('input[id^="edit-perm-"]:checked').forEach(checkbox => {
                permissions.push(checkbox.value);
            });
            
            // Find user index
            const userIndex = adminCredentials.findIndex(u => u.username === originalUsername);
            if (userIndex === -1) {
                alert('User not found');
                return;
            }
            
            // Update user data
            adminCredentials[userIndex].username = newUsername;
            adminCredentials[userIndex].role = newRole;
            adminCredentials[userIndex].permissions = permissions;
            
            // Update password if provided
            if (newPassword) {
                adminCredentials[userIndex].password = newPassword;
            }
            
            saveCredentials();
            
            // Update current user data in localStorage if it's the logged-in user
            const currentUserData = JSON.parse(localStorage.getItem('adminUserData') || '{}');
            if (currentUserData.username === originalUsername) {
                currentUserData.username = newUsername;
                currentUserData.role = newRole;
                currentUserData.permissions = permissions;
                if (newPassword) {
                    currentUserData.password = newPassword;
                }
                localStorage.setItem('adminUserData', JSON.stringify(currentUserData));
                
                // Update UI
                document.getElementById('username-display').textContent = newUsername;
                document.getElementById('username-display-mobile').textContent = newUsername;
                document.getElementById('current-user-role').textContent = newRole.toUpperCase();
                
                const badgeClass = `admin-badge-${newRole.toLowerCase().replace(' ', '-')}`;
                document.getElementById('user-role-badge').textContent = newRole.toUpperCase();
                document.getElementById('user-role-badge').className = `admin-badge ${badgeClass}`;
                document.getElementById('user-role-badge-mobile').textContent = newRole.toUpperCase();
                document.getElementById('user-role-badge-mobile').className = `admin-badge ${badgeClass}`;
                
                // Re-apply permissions
                applyUserPermissions(currentUserData);
            }
            
            // Refresh user management table
            populateUserManagementTable();
            
            // Close modal
            closeEditUserModal();
            
            alert('User updated successfully!');
        }

        // Enhanced add new user function
        function addNewUser() {
            const newUsername = prompt('Enter new username:');
            if (!newUsername) return;
            
            const newPassword = prompt('Enter new password:');
            if (!newPassword) return;
            
            if (newPassword.length < 6) {
                alert('Password must be at least 6 characters long');
                return;
            }
            
            const newRole = prompt('Enter role (manager, admin, technician, Sales Person):');
            if (!newRole) return;
            
            // Check if username already exists
            if (adminCredentials.find(u => u.username === newUsername)) {
                alert('Username already exists');
                return;

            // Default permissions based on role
            let defaultPermissions = ["inventory", "sales"];
            if (newRole === "manager") {
                defaultPermissions = ["inventory", "sales", "settings", "users", "products", "manager"];

                defaultPermissions = ["inventory", "sales", "products"];
            }
            
            // Add new user
            adminCredentials.push({
                username: newUsername,
                password: newPassword,
                role: newRole,
                permissions: defaultPermissions
            });
            
            saveCredentials();
            populateUserManagementTable();
            alert(`User ${newUsername} added successfully!`);
        }

        // Check if we need to reset daily sales
        function checkForDailyRefresh() {
            const today = new Date().toDateString();
            const lastRefresh = localStorage.getItem('lastDailyRefresh');
            
            // If it's a new day, reset daily sales
            if (lastRefresh !== today) {
                resetDailySales();
                localStorage.setItem('lastDailyRefresh', today);
            }
        }

        // Reset daily sales data
        function resetDailySales() {
            // Get the current sales data from localStorage
            const savedProducts = localStorage.getItem('elymax-products');
            if (savedProducts) {
                const parsed = JSON.parse(savedProducts);
                
                // Reset daily sales for each product
                parsed.forEach(product => {
                    product.dailySales = 0;
                });
                
                // Save back to localStorage
                localStorage.setItem('elymax-products', JSON.stringify(parsed));
                
                // Update the products array
                parsed.forEach(savedProduct => {
                    const index = products.findIndex(p => p.id === savedProduct.id);
                    if (index !== -1) {
                        products[index].dailySales = 0;
                    }
                });
                
                console.log('Daily sales reset for new day');
            }
        }

        // Apply user-specific permissions
        function applyUserPermissions(user) {
            // Get all permission-based elements
            const permissionElements = document.querySelectorAll('[data-permission]');
            
            permissionElements.forEach(element => {
                const requiredPermission = element.getAttribute('data-permission');
                const hasPermission = user.permissions.includes(requiredPermission);

        // Place order
        function placeOrder() {
            // Validate customer information
            const customerName = document.getElementById('customer-name').value.trim();
            const customerPhone = document.getElementById('customer-phone').value.trim();
            const customerAddress = document.getElementById('customer-address').value.trim();
            const customerCity = document.getElementById('customer-city').value.trim();
            const shippingMethod = document.getElementById('shipping-method').value;
            
            if (!customerName || !customerPhone || !customerAddress || !customerCity || !shippingMethod) {
                alert('Please fill in all required customer information fields.');
                return;
            }
            
            // Validate phone number format
            const phoneRegex = /^(\+254|0)[17]\d{8}$/;
            if (!phoneRegex.test(customerPhone.replace(/\s/g, ''))) {
                alert('Please enter a valid Kenyan phone number (e.g., 0712 345 678 or +254712345678)');
                return;
            }
            
            // Check stock availability
            for (const item of shoppingCart) {
                const product = products.find(p => p.id === item.id);
                if (!product || product.stock < item.quantity) {
                    alert(`Sorry, ${item.name} is out of stock or doesn't have enough quantity available.`);
                    return;
                }
            }
            
            // Process the order
            const orderId = `#ELYMX-${Date.now()}`;
            document.getElementById('order-id').textContent = orderId;
            
            // Update product stock and sales
            shoppingCart.forEach(item => {
                const product = products.find(p => p.id === item.id);
                if (product) {
                    product.stock -= item.quantity;
                    product.sold += item.quantity;
                    
                    // Update sales tracking
                    const now = new Date();
                    if (!product.dailySales) product.dailySales = 0;
                    if (!product.weeklySales) product.weeklySales = 0;
                    if (!product.monthlySales) product.monthlySales = 0;
                    if (!product.yearlySales) product.yearlySales = 0;
                    
                    product.dailySales += item.quantity;
                    product.weeklySales += item.quantity;
                    product.monthlySales += item.quantity;
                    product.yearlySales += item.quantity;
                    product.lastSaleDate = now;
                    
                    // Add to sales history
                    for (let i = 0; i < item.quantity; i++) {
                        salesHistory.push({
                            id: Date.now() + i,
                            productId: product.id,
                            productName: product.name,
                            productPrice: product.price,
                            quantity: 1,
                            total: product.price,
                            timestamp: now,
                            date: now.toDateString()
                        });
                    }
                }
            });
            
            // Prepare order details for WhatsApp notification
            const orderDetails = {
                orderId,
                customerName,
                customerPhone,
                customerAddress,
                customerCity,
                shippingMethod,
                items: shoppingCart.map(item => ({
                    name: item.name,
                    quantity: item.quantity,
                    price: item.price,
                    total: item.price * item.quantity
                })),
                total: document.getElementById('checkout-total').textContent
            };
            
            // Send WhatsApp notification
            sendWhatsAppNotification(orderDetails);
            
            // Save data
            saveToLocalStorage();
            saveSalesHistory();
            
            // Clear cart
            shoppingCart = [];
            saveCartToStorage();
            updateCartCount();
            
            // Close checkout modal and show confirmation
            closeCheckoutModal();
            document.getElementById('order-confirmation-modal').classList.remove('hidden');
            
            // Update product display
            renderProducts();
            updateInventoryStatus();
            
            // Reset customer form
            document.getElementById('customer-name').value = '';
            document.getElementById('customer-phone').value = '';
            document.getElementById('customer-address').value = '';
            document.getElementById('customer-city').value = '';
            document.getElementById('shipping-method').value = '';
        }

        // Sell a product (reduce stock, increase sold count) - For single sales
        function sellProduct(productId) {
            const product = products.find(p => p.id === productId);
            if (product && product.stock > 0) {
                product.stock--;
                product.sold++;
                
                // Update sales data for different periods
                const now = new Date();
                const today = now.toDateString();
                
                // Initialize sales tracking if not present
                if (!product.dailySales) product.dailySales = 0;
                if (!product.weeklySales) product.weeklySales = 0;
                if (!product.monthlySales) product.monthlySales = 0;
                if (!product.yearlySales) product.yearlySales = 0;
                
                // Update daily sales
                product.dailySales++;
                
                // Update weekly sales (if this week)
                const oneWeekAgo = new Date();
                oneWeekAgo.setDate(oneWeekAgo.getDate() - 7);
                if (new Date(product.lastSaleDate || now) > oneWeekAgo) {
                    product.weeklySales++;
                } else {
                    product.weeklySales = 1; // Reset if it's a new week
                }
                
                // Update monthly sales (if this month)
                const oneMonthAgo = new Date();
                oneMonthAgo.setMonth(oneMonthAgo.getMonth() - 1);
                if (new Date(product.lastSaleDate || now) > oneMonthAgo) {
                    product.monthlySales++;
                } else {
                    product.monthlySales = 1; // Reset if it's a new month
                }
                
                // Update yearly sales (if this year)
                const oneYearAgo = new Date();
                oneYearAgo.setFullYear(oneYearAgo.getFullYear() - 1);
                if (new Date(product.lastSaleDate || now) > oneYearAgo) {
                    product.yearlySales++;
                } else {
                    product.yearlySales = 1; // Reset if it's a new year
                }
                
                // Update last sale date
                product.lastSaleDate = now;
                
                // Add to sales history
                salesHistory.push({
                    id: Date.now(),
                    productId: product.id,
                    productName: product.name,
                    productPrice: product.price,
                    quantity: 1,
                    total: product.price,
                    timestamp: now,
                    date: today
                });
                
                // Update the display
                renderProducts();
                updateInventoryStatus();
                
                // Show confirmation
                alert(`Sold 1 unit of ${product.name}!`);
                
                // Save to localStorage
                saveToLocalStorage();
                saveSalesHistory();
            }
        }

            tableBody.innerHTML = '';

            products.forEach(product => {

                    case 'daily':
                        unitsSold = product.dailySales || 0;
                        break;

                        unitsSold = product.weeklySales || 0;
                        break;
                    case 'monthly':
                        unitsSold = product.monthlySales || 0;
                        break;

                        unitsSold = product.yearlySales || 0;
                        break;

                        unitsSold = product.sold || 0;

                totalUnits += unitsSold;

        // Update category sales
        function updateCategorySales(period) {
            const categorySales = document.getElementById('category-sales');
            categorySales.innerHTML = '';
            
            const categories = {};
            
            products.forEach(product => {
                let unitsSold = 0;
                
                switch(period) {
                    case 'daily':
                        unitsSold = product.dailySales || 0;
                        break;
                    case 'weekly':
                        unitsSold = product.weeklySales || 0;
                        break;
                    case 'monthly':
                        unitsSold = product.monthlySales || 0;
                        break;
                    case 'yearly':
                        unitsSold = product.yearlySales || 0;
                        break;
                    default:
                        unitsSold = product.sold || 0;
                }
                
                if (unitsSold > 0) {
                    if (!categories[product.category]) {
                        categories[product.category] = {
                            units: 0,
                            revenue: 0
                        };
                    }
                    
                    categories[product.category].units += unitsSold;
                    categories[product.category].revenue += product.price * unitsSold;
                }
            });
            
            if (Object.keys(categories).length === 0) {
                categorySales.innerHTML = `
                    <div class="empty-state">
                        <i class="fas fa-folder"></i>
                        <p>No category sales data</p>
                    </div>
                `;
                return;
            }
            
            Object.entries(categories).forEach(([category, data]) => {
                const categoryItem = document.createElement('div');
                categoryItem.className = 'flex justify-between items-center py-2 border-b border-gray-100';
                categoryItem.innerHTML = `
                    <span class="font-medium">${category}</span>
                    <div class="text-right">
                        <div class="font-semibold">${data.units} units</div>
                        <div class="text-sm text-gray-600">KSh ${data.revenue.toLocaleString()}</div>
                    </div>
                `;
                categorySales.appendChild(categoryItem);
            });
        }

        // Update date range display based on period
        function updateDateRangeDisplay(period) {
            const display = document.getElementById('date-range-display');
            const now = new Date();
            
            let rangeText = '';
            
            switch(period) {
                case 'daily':
                    rangeText = `Today: ${now.toLocaleDateString()}`;
                    break;
                case 'weekly':
                    const weekStart = new Date(now);
                    weekStart.setDate(now.getDate() - now.getDay());
                    rangeText = `This Week: ${weekStart.toLocaleDateString()} - ${now.toLocaleDateString()}`;
                    break;
                case 'monthly':
                    const monthStart = new Date(now.getFullYear(), now.getMonth(), 1);
                    rangeText = `This Month: ${monthStart.toLocaleDateString()} - ${now.toLocaleDateString()}`;
                    break;
                case 'yearly':
                    const yearStart = new Date(now.getFullYear(), 0, 1);
                    rangeText = `This Year: ${yearStart.toLocaleDateString()} - ${now.toLocaleDateString()}`;
                    break;
                case 'custom':
                    const startDate = document.getElementById('start-date').value;
                    const endDate = document.getElementById('end-date').value;
                    if (startDate && endDate) {
                        rangeText = `Custom Range: ${startDate} - ${endDate}`;
                    } else {
                        rangeText = 'Custom Range: Please select dates';
                    }
                    break;
            }
            
            display.textContent = rangeText;
        }

            const savedProducts = localStorage.getItem('elymax-products');
            if (savedProducts) {
                const parsed = JSON.parse(savedProducts);
                // Update products array with saved data
                parsed.forEach(savedProduct => {
                    const index = products.findIndex(p => p.id === savedProduct.id);
                    if (index !== -1) {
                        products[index].stock = savedProduct.stock;
                        products[index].sold = savedProduct.sold;
                        products[index].dailySales = savedProduct.dailySales || 0;
                        products[index].weeklySales = savedProduct.weeklySales || 0;
                        products[index].monthlySales = savedProduct.monthlySales || 0;
                        products[index].yearlySales = savedProduct.yearlySales || 0;
                        products[index].lastSaleDate = savedProduct.lastSaleDate ? new Date(savedProduct.lastSaleDate) : null;
                    }
                });
            }
